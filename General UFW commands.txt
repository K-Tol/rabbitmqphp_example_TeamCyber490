sudo apt update
sudo apt install -y ufw
sudo ufw --force reset

//UFW comes pre installed already, but use this if it is not on VM
//or if any update, reset is needed 


//default/deny outbound + inbound *


sudo ufw default deny incoming
sudo ufw default allow outgoing



//allow SSH only over Tailscale *

sudo ufw allow in on tailscale0 to any port 22 proto tcp


//enable and status check


sudo ufw --force enable
sudo ufw status verbose

// to delete

sudo ufw status verbose

// to disable

sudo ufw disable

WEB VM:
sudo ufw default deny incoming
sudo ufw default allow outgoing

//Only allow HTTP/HTTPS on tailscale , port 80 and 443

sudo ufw allow in on tailscale0 to any port 80 proto tcp

sudo ufw allow in on tailscale0 to any port 443 proto tcp

To Deny public HTTP/HTTPS traffic

sudo ufw deny 80/tcp
sudo ufw deny 443/tcp

BROKER VM: 

sudo ufw allow in on tailscale0 to any port 5672 proto tcp
sudo ufw deny 5672/tcp
